```python
import cvxpy as cp
import networkx as nx
import numpy as np

L = [['zhang','li','wang'],['li','zhao','liu'],['zhang','liu','wang'],['zhao','liu','sun'],['zhang','wangg','sun'],['li','liu','wang']]


w = np.zeros((6,6))
for i in range(5): #取0-4
	for j in range(i+1,6):   #对矩阵上三角
		if len(set(L[i]) & set(L[j])) >=1:#列表转为集合，如果有任意重复元素重合，则交集元素个数大于一
			w[i,j]=1
			
ni,nj=np.nonzero(w) #nonzero()返回行列信息
w = w+w.T #将w改回对称的临接矩阵

deg = w.sum(axis=1) #将1相加，求各个顶点的度
K = int(max(deg))#得到最大度
n = len(w)#顶点的个数

#定义变量
x=cp.Variable((n,K+1))
y=cp.Variable()

#目标函数
obj=cp.Minimize(y)
#约束函数
con=[cp.sum(x,axis=1)==1,x>=0,x<=1]
for i in range(n):
	con.append(y>=range(1,K+2)@x[i,:])
for k in range (K+1):
	for i in range (len(ni)):
		con.append(x[ni[i],k]+x[nj[i],k]<=1)
		
		
#求解
prob = cp.Problem(obj, con)
prob.solve(solver='GLPK_MI')
i,k = np.nonzero(x.value)

print('最优值为：',prob.value)
print('最优解为：',x.value)
print('顶点与颜色的对于如下')
print("i=",i+1)
print("k=",k+1)
```
