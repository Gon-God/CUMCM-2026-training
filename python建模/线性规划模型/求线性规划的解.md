```python
import cvxpy as cp
from numpy import array
c = array([70,50,60]) #目标函数的系数
a = array([[2,4,3], [3,1,5],[7,3,5]]) #约束函数的系数矩阵
b = array([150,160,200]) #约束条件的右边向量
x = cp.Variable(1, pos=True) #定义三个决策变量
obj = cp.Maximize(c@x) #构造目标函数
cons = [a@x<=b] #构造约束条件
prob = cp.Problem(obj, cons) #封装函数的命令
prob.solve(solver = 'GLPK_MT')  #求解问题，GLPK_MT是一个开源的求解器
print('最优解为：',x.value)
print('最优值为：',prob.value)
```
