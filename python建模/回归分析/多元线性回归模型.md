```python
import numpy as np
import statsmodels.api as sm
import pylab as plt

a = np.array([
    [7.1, 11.1, 15.4],
    [6.8, 10.8, 15.0],
    [9.2, 12.4, 22.8],
    [11.4, 10.9, 27.8],
    [8.7, 9.6, 19.5],
    [6.6, 9.0, 13.1],
    [10.3, 10.5, 24.9],
    [10.6, 12.4, 26.2]
])

a=a.T


plt.plot(a[0],a[2],'*b',label = 'x1')
plt.plot(a[1],a[2],'og',label = 'x2')

d = {'x1':a[0],'x2':a[1],'y':a[2]}
re = sm.formula.ols('y~x1+x2',d).fit()
print(re.summary())
yh = re.predict({'x1':[9,10],'x2':[10,9]})

print('残差的方差：',re.mse_resid)
print('预测值：',yh)

plt.legend()
plt.show
```
残差的方差： 0.10207974705778973
预测值： 
0    20.591832
1    22.900808
