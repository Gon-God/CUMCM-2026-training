```python
import statsmodels.formula.api as smf
import numpy as np

# 假设数据存储在DataFrame 'df'中，有 'y' 和 'x' 两列
# 使用 'I(x**2)' 来添加 x 的平方项
model = smf.ols('y ~ x + I(x**2)', data=df).fit()

# 打印详细的统计摘要
print(model.summary())
```
