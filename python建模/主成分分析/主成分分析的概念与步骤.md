# 主成分分析的应用

`sklearn.decomposition.PCA(n_component = None)`

n_component = 2 表示取2个主成分
n_component = 0.85  表示提取累计贡献率达到85%

返回值是一个PCA对象
- `X_pca = pca.fit_transform(X_scaled)`可将其转为一个二维数组
-  **形状**：`(样本数, n_components)`
    
- **含义**：这就是原始数据降维后的新特征矩阵（主成分得分）。例如，你把 4 维数据降成 2 维，返回的就是一个 150 行 2 列的数组。


```python

import numpy as np

import pandas as pd

from sklearn.decomposition import PCA

from sklearn.datasets import load_iris

from sklearn.preprocessing import StandardScaler

import matplotlib.pyplot as plt

  

# 加载数据

iris = load_iris()

X = iris.data  # 特征数据

y = iris.target # 标签（用于可视化）

  

scaler = StandardScaler() #标准化规则

X_scaled = scaler.fit_transform(X) #运用规则到数据上面

  

pca = PCA(n_components=2)

  

X_pca = pca.fit_transform(X_scaled) #运用规则到主成分降维

  

print(f"原始数据形状: {X.shape}")       # 输出: (150, 4)

print(f"降维后数据形状: {X_pca.shape}") # 输出: (150, 2)

plt.figure(figsize=(8, 6))

# 用不同颜色表示不同鸢尾花种类

scatter = plt.scatter(X_pca[:, 0], X_pca[:, 1], c=y, cmap='viridis', edgecolor='k', s=100)

plt.xlabel('主成分 1')

plt.ylabel('主成分 2')

plt.title('鸢尾花数据集的PCA降维结果')

plt.legend()

plt.show()
```
